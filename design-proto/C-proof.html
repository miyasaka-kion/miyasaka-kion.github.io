// sin(x) 的麦克劳林级数，截断到第 N 项
template <class T>
T sin_approx(T x, int N) {
T s = x, term = x;
for (int n = 1; n <= N; ++n) {
term *= -x*x / ((2*n) * (2*n+1));
s += term;
}
return s;
}
// 误差界由拉格朗日余项给出
